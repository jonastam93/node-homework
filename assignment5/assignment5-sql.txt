SELECT
    first_orders.order_id,
    SUM(products.price * line_items.quantity) AS total_price
FROM (
    SELECT order_id
    FROM orders
    ORDER BY order_id
    LIMIT 5
) AS first_orders
JOIN line_items
    ON first_orders.order_id = line_items.order_id
JOIN products
    ON line_items.product_id = products.product_id
GROUP BY first_orders.order_id
ORDER BY first_orders.order_id;

SELECT
    c.customer_name,
    AVG(t.total_price) AS average_order_price
FROM customers AS c
JOIN (
    SELECT
        orders.order_id,
        orders.customer_id,
        SUM(products.price * line_items.quantity) AS total_price
    FROM orders
    JOIN line_items
        ON orders.order_id = line_items.order_id
    JOIN products
        ON line_items.product_id = products.product_id
    GROUP BY
        orders.order_id,
        orders.customer_id
) AS t
    ON c.customer_id = t.customer_id
GROUP BY
    c.customer_id,
    c.customer_name
ORDER BY
    c.customer_name;

SELECT
    employees.first_name,
    employees.last_name,
    COUNT(orders.order_id) AS order_count
FROM employees
JOIN orders
    ON employees.employee_id = orders.employee_id
GROUP BY
    employees.employee_id,
    employees.first_name,
    employees.last_name
HAVING COUNT(orders.order_id) > 5
ORDER BY employees.last_name;
